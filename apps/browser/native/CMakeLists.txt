cmake_minimum_required(VERSION 3.21)
project(GirruBrowser LANGUAGES C CXX OBJCXX)
set(CMAKE_CXX_STANDARD 20)
set(CEF_ROOT "${CEF_ROOT}" CACHE PATH "Verified CEF binary SDK")
list(APPEND CMAKE_MODULE_PATH "${CEF_ROOT}/cmake")
find_package(CEF REQUIRED)
add_subdirectory("${CEF_LIBCEF_DLL_WRAPPER_PATH}" wrapper)
add_executable(GirruBrowser MACOSX_BUNDLE main_mac.mm prompts_mac.mm browser.cc)
SET_EXECUTABLE_TARGET_PROPERTIES(GirruBrowser)
target_include_directories(GirruBrowser PRIVATE "${CEF_ROOT}" "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(GirruBrowser libcef_dll_wrapper ${CEF_STANDARD_LIBS})
set_target_properties(GirruBrowser PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist.in")
add_executable(GirruBrowserHelper MACOSX_BUNDLE helper.cc)
SET_EXECUTABLE_TARGET_PROPERTIES(GirruBrowserHelper)
target_include_directories(GirruBrowserHelper PRIVATE "${CEF_ROOT}")
target_link_libraries(GirruBrowserHelper libcef_dll_wrapper ${CEF_STANDARD_LIBS})
set_target_properties(GirruBrowserHelper PROPERTIES OUTPUT_NAME "GirruBrowser Helper" MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Helper.plist.in")
add_dependencies(GirruBrowser GirruBrowserHelper)
